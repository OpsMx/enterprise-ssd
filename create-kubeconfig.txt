#!/bin/bash

echo "Plase Enter the Service-Account-Name:"
read SERVICE_ACCOUNT_NAME
echo "Plase Enter the NameSpace:"
read NAMESPACE
echo "Plase Enter the Access Type Of Kube File:(Readonly | Admin)"
read AccessType

#SERVICE_ACCOUNT_NAME=phani
#NAMESPACE=default
server=$(kubectl config view --minify --output jsonpath='{.clusters[*].cluster.server}')
context=$(kubectl config current-context)
name=$(kubectl config get-contexts "$context" | awk '{print $3}' | tail -n 1)
user=${SERVICE_ACCOUNT_NAME}-${NAMESPACE}-${name}

kubectl create serviceaccount ${SERVICE_ACCOUNT_NAME}-sa -n ${NAMESPACE}

kubectl create token ${SERVICE_ACCOUNT_NAME}-sa -n ${NAMESPACE}

cat << EOF | kubectl apply -f -
apiVersion: v1
kind: Secret
type: kubernetes.io/service-account-token
metadata:
   name: ${SERVICE_ACCOUNT_NAME}
   annotations:
      kubernetes.io/service-account.name: "${SERVICE_ACCOUNT_NAME}-sa"
EOF

if [ ${AccessType} = "Admin" ] ;
then
kubectl create clusterrolebinding ${SERVICE_ACCOUNT_NAME}-rolebinding --clusterrole=cluster-admin --serviceaccount=${NAMESPACE}:${SERVICE_ACCOUNT_NAME}-sa
kubeconfigname=${SERVICE_ACCOUNT_NAME}.config
elif [ ${AccessType} = "Readonly" ] ;
then
cat <<EOF | kubectl apply -f -
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRole
metadata:
  annotations:
    rbac.authorization.kubernetes.io/autoupdate: "true"
  labels:
  name: opsmx-readonly
  namespace: ${NAMESPACE}
rules:
  - apiGroups:
     - metrics.k8s.io
    resources:
     - pods
     - nodes
    verbs:
      - get
      - list
      - watch
  - apiGroups:
      - ""
    resources: ["*"]
    verbs:
      - get
      - list
      - watch
  - apiGroups:
      - extensions
    resources: ["*"]
    verbs:
      - get
      - list
      - watch
  - apiGroups:
      - apps
    resources: ["*"]
    verbs:
      - get
      - list
      - watch
---
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: opsmx-readonly-binding
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: opsmx-readonly
subjects:
  - kind: ServiceAccount
    name: ${SERVICE_ACCOUNT_NAME}-sa
    namespace: ${NAMESPACE}
EOF
kubeconfigname=${SERVICE_ACCOUNT_NAME}-readonly.config
user=${user}-readonly
else
echo "Accesstype not valid please check and run again !!..";
fi

ca=$(kubectl get secret ${SERVICE_ACCOUNT_NAME} -n ${NAMESPACE} -o jsonpath='{.data.ca\.crt}')

token=$(kubectl get secret ${SERVICE_ACCOUNT_NAME}  -n ${NAMESPACE} -o jsonpath='{.data.token}' | base64 --decode)

echo "
apiVersion: v1
kind: Config
clusters:
- name: ${name}
  cluster:
    certificate-authority-data: ${ca}
    server: ${server}
contexts:
- name: ${user}-context
  context:
    cluster: ${name}
    namespace: ${NAMESPACE}
    user: ${user}
current-context: ${user}-context
users:
- name: ${user}
  user:
    token: ${token}
" > ${kubeconfigname}
echo "${kubeconfigname} generated successfully .."
